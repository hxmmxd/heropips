Python sidecar.py (in Wine)
@app.post("/trade")
def execute_trade(payload: TradePayload):
    price = mt5.symbol_info_tick(payload.symbol).ask
    request = {
        "action": mt5.TRADE_ACTION_DEAL,
        "symbol": payload.symbol,
        "volume": payload.volume,
        "type": mt5.ORDER_TYPE_BUY,
        "price": price,
        "sl": payload.stopLoss,
        "tp": payload.takeProfit,
        "deviation": 20,
        "type_filling": mt5.ORDER_FILLING_IOC
    }
    res = mt5.order_send(request)
    return {"orderId": res.order, "openPrice": res.price}
Dockerfile base configuration
FROM ubuntu:22.04
RUN dpkg --add-architecture i386 && apt-get update \
    && apt-get install -y winehq-stable xvfb redsocks \
    && rm -rf /var/lib/apt/lists/*
RUN wine python-installer.exe /quiet InstallAllUsers=1
RUN wine python -m pip install MetaTrader5 fastapi uvicorn
COPY entrypoint.sh /home/trader/entrypoint.sh
EXPOSE 8000
ENTRYPOINT ["/home/trader/entrypoint.sh"]
